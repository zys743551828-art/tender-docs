IMAGE = "sha256:620f97f10e54173f1bcc574168f4626a3d07143aecc91ae038b70f733369fe59"

from deploy.disposable_docker import DockerTestRun

def main():
    OUT.mkdir(parents=True)
    (OUT / "runner.py.txt").write_bytes(Path(__file__).read_bytes())
    from deploy.word_execution_host import clean_environment

    environment = clean_environment()
    environment.update(DOCKER_HOST="npipe:////./pipe/dockerDesktopLinuxEngine", WORKBENCH_ENV_FILE="explicit-host-bind-regression", PYTHONDONTWRITEBYTECODE="1")
    os.environ.clear();os.environ.update(environment)
    code = 1
    try:
        with DockerTestRun(OUT / "resources.json") as run:
            mounts = []
            for name in ("packages", "deploy", "tests", "pyproject.toml", "requirements.lock.txt"):
                mounts += ["--mount", f"type=bind,source={SOURCE/name},target=/source/{name},readonly"]
            identity = run.container(IMAGE, ["--network","none","--ipc","none","--read-only","--memory","1g","--cpus","1","--pids-limit","128","--user","65532:65532","--cap-drop","ALL","--security-opt","no-new-privileges:true","--tmpfs","/tmp:rw,nosuid,nodev,size=512m,mode=1777","--mount",f"type=bind,source={OUT},target=/evidence",*mounts,"--env","WORKBENCH_ENV_FILE=explicit-host-bind-regression","--env","PYTHONDONTWRITEBYTECODE=1","--env","PYTEST_DISABLE_PLUGIN_AUTOLOAD=1","--workdir","/source","--entrypoint","python"], ["-B","-c",DRIVER])
            result=run._call('start','--attach',identity,timeout=300)
            (OUT/'driver.log').write_text(result.stdout+'\n'+result.stderr,encoding='utf-8')
            print(result.stdout,flush=True);print(result.stderr,flush=True);code=result.returncode
    finally:
        receipt=json.loads((OUT/'resources.json').read_text())
        print(json.dumps({'evidence':str(OUT),'exit_code':code,'cleanup_closed':receipt['closed']}),flush=True)
    return code


if __name__ == '__main__':
    raise SystemExit(main())
