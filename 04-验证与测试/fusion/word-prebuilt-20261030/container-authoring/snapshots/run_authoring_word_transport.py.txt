"""Run-owned real PAGE/RICH mailbox round trip; no DB, model or business socket."""

from __future__ import annotations

import datetime
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile
import time
import traceback

ROOT = Path(__file__).resolve().parent
APP = ROOT / "tender-app-dev/workbench"
IMAGE = "tender-workbench-runtime:py3.12.16-candidate-20261030"
os.environ["WORKBENCH_ENV_FILE"] = "explicit-authoring-transport-no-business-env"
os.environ["PYTHONDONTWRITEBYTECODE"] = "1"
sys.path[:0] = [str(APP), str(APP / "tests")]

import pypdfium2
import test_authoring_generated as page_cases
import test_authoring_rich_renderer as rich_cases

from deploy.disposable_docker import DockerTestRun
from packages.authoring import generated_docx, prebuilt_generated, prebuilt_rich
from packages.word_intake import execution_transport as codec
from packages.word_intake import prebuilt


class LocalRun(DockerTestRun):
    @staticmethod
    def _call(*args, timeout=60, env=None):
        value = prebuilt._call(list(args), timeout=timeout)
        return subprocess.CompletedProcess(
            value.args, value.returncode, value.stdout.decode("utf-8", errors="replace"),
            value.stderr.decode("utf-8", errors="replace"),
        )


CLIENT = r'''
import json,os,sys
from pathlib import Path
from packages.word_intake import execution_transport as codec
lane=sys.argv[1]
assert lane in {'page','rich'}
assert os.geteuid()==65532
raw=Path('/fixtures',lane+'.request.json').read_bytes()
request=codec.decode_request(raw)
assert request['lane']==lane
result=codec.call(lane,**request['arguments'])
assert result['resource_receipt']['closed'] is True
sys.stdout.buffer.write(codec.encode_response(raw,result))
'''


def safe(value):
    if isinstance(value, bytes):
        return {"byte_size": len(value), "sha256": hashlib.sha256(value).hexdigest()}
    if isinstance(value, dict):
        return {k: safe(v) for k, v in value.items()}
    if isinstance(value, list):
        return [safe(v) for v in value]
    return value


def record(out, lane, result, source):
    (out / f"{lane}.docx").write_bytes(source)
    (out / f"{lane}.json").write_text(json.dumps(safe(result), ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    if result.get("pdf"):
        (out / f"{lane}.pdf").write_bytes(result["pdf"])
        document = pypdfium2.PdfDocument(result["pdf"])
        try:
            for index in range(min(len(document), 6)):
                page = document[index]
                bitmap = page.render(scale=1.4)
                pixels = bitmap.to_pil()
                try:
                    pixels.save(out / f"{lane}-{index + 1}.png")
                finally:
                    pixels.close()
                    bitmap.close()
                    page.close()
        finally:
            document.close()
    for index, iteration in enumerate(result.get("render_iterations", []), 1):
        (out / f"{lane}-pass-{index}.docx").write_bytes(iteration["docx_bytes"])
        (out / f"{lane}-pass-{index}.pdf").write_bytes(iteration["pdf_bytes"])


def main():
    out = ROOT / "evidence-local-20261030" / ("authoring-word-transport-" + datetime.datetime.now().strftime("%Y%m%d-%H%M%S"))
    out.mkdir()
    fixtures = out / "fixtures"
    fixtures.mkdir()
    scratch = out / "operator-tmp"
    scratch.mkdir()
    os.environ["TEMP"] = os.environ["TMP"] = str(scratch)
    tempfile.tempdir = str(scratch)
    status = {"state": "starting", "stage": "prepare", "model_calls": 0, "database_connections": 0,
              "client_image_reference": IMAGE, "evidence": str(out),
              "started_at_utc": datetime.datetime.now(datetime.timezone.utc).isoformat()}
    run = LocalRun(out / "client-resources.json")
    host = None
    stop = out / "operator-stop"
    exit_code = 1
    try:
        source_names = [
            "deploy/word_execution_host.py", "deploy/Dockerfile.runtime", "deploy/fusion_full_requirements.txt",
            "requirements.lock.txt", "packages/word_intake/execution_transport.py", "packages/word_intake/prebuilt.py",
            "packages/authoring/prebuilt_generated.py", "packages/authoring/prebuilt_rich.py",
            "packages/authoring/prebuilt_pdf.py", "packages/authoring/rich_caption_pdf.py",
            "packages/authoring/rich_table_pdf.py",
            "tests/test_authoring_generated.py", "tests/test_authoring_rich_renderer.py", "tests/test_authoring_rich.py",
        ]
        (out / "source-correspondence.json").write_text(json.dumps({
            "files": {name: hashlib.sha256((APP / name).read_bytes()).hexdigest() for name in source_names},
            "driver_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        }, indent=2) + "\n", encoding="utf-8")
        provider = prebuilt.build_profile(receipt=out / "profile-resources.json")
        snapshot = page_cases.snapshot.__wrapped__()
        source = generated_docx.assemble(snapshot)
        requests = {"page": codec._request("page", {
            "source": source, "snapshot": snapshot,
            "producer_manifest": generated_docx.producer_manifest(source, snapshot),
            "profile": prebuilt_generated.profile_for(provider),
        })}
        rich_snapshot, materials, _, pagination, _ = rich_cases.rich_input.__wrapped__()
        requests["rich"] = codec._request("rich", {
            "snapshot": rich_snapshot, "materials": materials,
            "profile": prebuilt_rich.profile_for(provider), "mode": pagination["mode"],
        })
        for lane, raw in requests.items():
            (fixtures / f"{lane}.request.json").write_bytes(raw)
        (out / "provider-profile.json").write_text(json.dumps(provider, indent=2) + "\n", encoding="utf-8")
        image = json.loads(prebuilt._successful(["image", "inspect", IMAGE]))[0]
        status["client_image_id"] = image["Id"]
        with run:
            identity = run.container(image["Id"], [
                "--network", "none", "--ipc", "none", "--user", "65532:65532", "--read-only",
                "--cap-drop", "ALL", "--security-opt", "no-new-privileges", "--memory", "1g",
                "--cpus", "1", "--pids-limit", "128", "--tmpfs", "/tmp:rw,nosuid,nodev,size=512m,mode=1777",
                "--mount", f"type=bind,source={APP / 'packages'},target=/workspace/packages,readonly",
                "--mount", f"type=bind,source={fixtures},target=/fixtures,readonly",
                "--env", "PYTHONPATH=/workspace", "--env", "PYTHONDONTWRITEBYTECODE=1",
                "--env", "WORKBENCH_ENV_FILE=explicit-linux-transport-no-business-env",
                "--env", "WORD_RENDER_EXECUTION_DIRECTORY=/tmp/word-execution",
                "--entrypoint", "python",
            ], ["-B", "-c", "import time;time.sleep(900)"])
            run.start(identity)
            observed = json.loads(prebuilt._successful(["container", "inspect", identity]))[0]
            assert observed["HostConfig"]["NetworkMode"] == "none"
            assert observed["HostConfig"]["ReadonlyRootfs"] is True
            assert observed["Config"]["User"] == "65532:65532"
            assert not observed["HostConfig"]["Privileged"]
            assert all("docker.sock" not in str(item) and "docker_engine" not in str(item) for item in observed["Mounts"])
            kernel = json.loads(prebuilt._successful([
                "exec", identity, "python", "-I", "-c",
                "import json,os;from pathlib import Path;rows=[]\n"
                "for line in Path('/proc/self/mountinfo').read_text().splitlines():\n"
                " before,sep,after=line.partition(' - ');fields=before.split()\n"
                " if sep and fields[4]=='/tmp':rows.append({'mount_point':fields[4],'filesystem':after.split()[0],'options':fields[5]})\n"
                "print(json.dumps({'uid':os.geteuid(),'tmp_mounts':rows}))",
            ]))
            assert kernel["uid"] == 65532 and len(kernel["tmp_mounts"]) == 1
            assert kernel["tmp_mounts"][0]["filesystem"] == "tmpfs"
            status["client"] = {
                "id": identity, "run_id": run.run_id, "user": observed["Config"]["User"],
                "image_id": observed["Image"], "network_mode": observed["HostConfig"]["NetworkMode"],
                "read_only_root": observed["HostConfig"]["ReadonlyRootfs"],
                "ipc_mode": observed["HostConfig"]["IpcMode"], "mounts": observed["Mounts"],
                "configured_tmpfs": observed["HostConfig"]["Tmpfs"], "kernel": kernel,
                "cap_drop": observed["HostConfig"]["CapDrop"], "security_opt": observed["HostConfig"]["SecurityOpt"],
            }
            status["stage"] = "host-start"
            environment = prebuilt._environment() | {
                "WORKBENCH_ENV_FILE": "explicit-host-transport-no-business-env", "PYTHONDONTWRITEBYTECODE": "1",
            }
            with (out / "host.stdout.json").open("wb") as stdout, (out / "host.stderr.log").open("wb") as stderr:
                host = subprocess.Popen([
                    sys.executable, "-B", str(APP / "deploy/word_execution_host.py"),
                    "--container-id", identity, "--run-id", run.run_id,
                    "--report-dir", str(out / "host-private"), "--stop-file", str(stop),
                ], cwd=APP, env=environment, stdout=stdout, stderr=stderr,
                    creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0))
                try:
                    ready = out / "host-private/ready.json"
                    ready_deadline = time.monotonic() + 30
                    while not ready.exists():
                        if host.poll() is not None or time.monotonic() >= ready_deadline:
                            raise RuntimeError("HOST_EXECUTOR_NOT_READY")
                        time.sleep(0.05)
                    assert json.loads(ready.read_text(encoding="utf-8")) == {
                        "state": "ready", "container_id": identity, "run_id": run.run_id,
                    }
                    for lane, request_raw in requests.items():
                        status["stage"] = lane
                        if host.poll() is not None:
                            raise RuntimeError("HOST_EXECUTOR_EXITED_BEFORE_CLIENT")
                        print(json.dumps({"evidence": str(out), "stage": lane}), flush=True)
                        execution = prebuilt._call([
                            "exec", "-i", identity, "python", "-B", "-c", CLIENT, lane,
                        ], timeout=300)
                        (out / f"{lane}-client.stderr.log").write_bytes(execution.stderr)
                        if execution.returncode:
                            raise RuntimeError(f"{lane.upper()}_CLIENT_FAILED")
                        result = codec.decode_response(request_raw, execution.stdout)
                        record(out, lane, result, source if lane == "page" else result["docx"])
                        status[lane] = {key: result.get(key) for key in ("state", "coverage", "gaps", "error_code")}
                        assert result["state"] == "succeeded" and result["coverage"] == "complete" and result["gaps"] == []
                        if lane == "rich":
                            assert result["navigation_proof"]["fixed_point"] is True
                            assert len(result["render_iterations"]) >= 2
                finally:
                    stop.write_text("stop after bounded current job and cleanup\n", encoding="utf-8")
                    host.wait(timeout=300)
                status["host_exit_code"] = host.returncode
                host_summary = json.loads((out / "host-private/summary.json").read_text(encoding="utf-8"))
                status["host"] = host_summary
                assert host.returncode == 0 and host_summary["cleanup_closed"] and host_summary["processed"] == 2
                assert run._owned(run.resources[0])["State"]["Running"] is True
                status["state"] = "passed"
                status["stage"] = "complete"
                exit_code = 0
    except Exception as error:
        status.update(state="failed", error_type=type(error).__name__, error_code=str(error) if isinstance(error, RuntimeError) else "ASSERTION_OR_EXECUTION_FAILED")
        (out / "driver-error.log").write_text(traceback.format_exc(), encoding="utf-8")
    finally:
        if not run.closed:
            try:
                run.cleanup()
            except Exception:
                status["client_cleanup_error"] = "CLIENT_CLEANUP_UNCONFIRMED"
        status["client_cleanup_closed"] = run.closed
        status["finished_at_utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
        (out / "summary.json").write_text(json.dumps(status, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
        print(json.dumps(status, ensure_ascii=False, indent=2), flush=True)
    return 0 if exit_code == 0 and status["state"] == "passed" and run.closed else 1


if __name__ == "__main__":
    raise SystemExit(main())
